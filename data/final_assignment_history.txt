def chunk_text(text, max_chars=800):
    chunks = []
    i = 0
    while i < len(text):
        j = min(i + max_chars, len(text))
        if j < len(text):
            sep = max(text.rfind(".", i, j),
                      text.rfind("\n", i, j),
                      text.rfind(" ", i, j))
            if sep > i:
                j = sep + 1
        c = text[i:j].strip()
        if len(c) > 20:
            chunks.append(c)
        i = j
    return chunks

"""

def chunk_text(text, max_chars=800):
    out, i, n = [], 0, len(text)
    while i < n:
        j = min(i + max_chars, n)
        if j < n:
            sep = max(text.rfind(".", i, j), text.rfind("\n", i, j), text.rfind(" ", i, j))
            if sep > i: j = sep + 1
        c = text[i:j].strip()

        i = j
    return out

    # OCR + embedder
    print("🔧 Loading OCR + embeddings...")
    ocr_reader = easyocr.Reader(["en"], gpu=False)
    embedder = SentenceTransformer(ARGS.embed)
    dim = embedder.get_sentence_embedding_dimension()

    os.makedirs(ARGS.images_dir, exist_ok=True)

    try:
        doc = fitz.open(ARGS.pdf)
    except Exception as e:
        print(f"⛔ Cannot open PDF: {e}")
        sys.exit(1)

    # Extract + collect
    print("📄 Extracting and chunking...")
    all_meta = []   # [{'type': 'text'|'image', 'page': int, 'content': str, 'caption': str?}]
    text_points = []  # list of text strings we will embed/index

    for p in range(len(doc)):
        page = doc.load_page(p)
        page_num = p + 1

        # 1) text (native or OCR)
        txt = page.get_text().strip()
        if not txt:
            arr = image_to_rgb_for_ocr(page)
            ocr = ocr_reader.readtext(arr)
            txt = " ".join([x[1] for x in ocr]).strip()

        if txt:
            for ch in chunk_text(txt):
                all_meta.append({"type": "text", "page": page_num, "content": ch})
                text_points.append(ch)

        # 2) inline images + (optional) captions (metadata only)
        for idx, img_meta in enumerate(page.get_images(full=True), start=1):
            xref = img_meta[0]
            img_dict = doc.extract_image(xref)
            ext = img_dict["ext"]

            cap = caption_image(path)

    print("🗄️ Building in-memory Qdrant index...")
    client = QdrantClient(":memory:")
    client.create_collection(
        collection_name=ARGS.collection,
        vectors_config=VectorParams(size=dim, distance=Distance.COSINE)
    )

    text_vectors = embedder.encode(text_points, show_progress_bar=False)
    pid = 0
    for i, vec in enumerate(text_vectors):
        # payload mirrors the text entry in all_meta (order preserved by construction)
        # But safer: scan all_meta to find the next text entry
        while pid < len(all_meta) and all_meta[pid]["type"] != "text":
            pid += 1
        payload = all_meta[pid]; pid += 1

        client.upsert(
            collection_name=ARGS.collection,
            points=[PointStruct(id=i, vector=vec.tolist(), payload=payload)]
        )

    print("✅ Index ready")

    # Load questions
    with open(ARGS.questions, "r", encoding="utf-8") as f:
        questions = [q.strip() for q in f if q.strip()]

    # Retrieval helper (text only + threshold)
    def retrieve_text_context(query, top_k):
        qv = embedder.encode(query).tolist()
        res = client.query_points(
            collection_name=ARGS.collection,
            query=qv,
            limit=top_k,
            with_payload=True,
            score_threshold=ARGS.min_score,
            # Filter text-only (but we only indexed text anyway)
        )
        chunks = []
        pages = set()
        for pt in res.points:
            c = pt.payload["content"]
            chunks.append(c)
            pages.add(pt.payload["page"])
        # attach up to 2 nearby image paths (same page or +/-1)
        images = []
        for pg in sorted(pages):
            for m in all_meta:
                if m["type"] == "image" and abs(m["page"] - pg) <= 1:
                    if m["content"] not in images:
                        images.append(m["content"])
                    if len(images) >= 2:
                        break
            if len(images) >= 2:
                break

        text_blob = "\n".join(chunks)
        limited = text_blob.encode("utf-8")[:ARGS.max_ctx_bytes].decode("utf-8", "ignore")
        return limited, images

    # Answer all questions
    with open(ARGS.answers, "w", encoding="utf-8") as out:
        print("🧠 Answering...")
        for i, q in enumerate(questions, 1):
            ctx, imgs = retrieve_text_context(q, ARGS.top_k)
            ans = generate_answer(ARGS.llama, q, ctx) if ctx.strip() else None
            out.write(f"Question {i}: {q}\n{ans or 'Not found in context.'}\n\n")

    print(f"✅ Answers saved to {ARGS.answers}")

    out, i, n = [], 0, len(text)
    while i < n:
        j = min(i + max_chars, n)
        if j < n:
            sep = max(text.rfind(".", i, j), text.rfind("\n", i, j), text.rfind(" ", i, j))

        i = j
    return out

# ------------------------- Helpers -------------------------
def chunk_text(text, max_chars=800):
    out, i, n = [], 0, len(text)
    while i < n:
        j = min(i + max_chars, n)
        if j < n:
            sep = max(text.rfind(".", i, j), text.rfind("\n", i, j), text.rfind(" ", i, j))
            if sep > i: j = sep + 1
        c = text[i:j].strip()
        if len(c) > 20: out.append(c)
        i = j
    return out

def page_to_rgb_array(page, dpi=OCR_DPI):
    pix = page.get_pixmap(dpi=dpi)
    mode = "RGBA" if pix.alpha else "RGB"
    img = Image.frombytes(mode, [pix.width, pix.height], pix.samples)
    if img.mode != "RGB":
        img = img.convert("RGB")
    return np.array(img)

def llava_caption(image_path, question_hint=None):
    if not USE_LLAVA_CAPTIONS:
        return ""
    base_prompt = "Describe this figure in one concise sentence."
    if question_hint:
        base_prompt += f" Focus on aspects relevant to: {question_hint}"

            capture_output=True, text=True, encoding="utf-8"
        )
        if proc.returncode != 0:
            return ""
        return proc.stdout.strip()
    except Exception:
        return ""

def trim_to_limit(s, limit=300):
    if len(s) <= limit:
        return s
    cut = s[:limit]
    # try to end on sentence/word boundary
    for p in [".", "!", "?", ";", ":"]:
        idx = cut.rfind(p)
        if idx >= int(limit*0.6):
            return cut[:idx+1].strip()
    # else soft cut at last space
    sp = cut.rfind(" ")
    return (cut[:sp] if sp > 0 else cut).strip()

def extract_chunks(pdf_path):
    doc = fitz.open(pdf_path)
    meta = []  # list of dicts: {"type":"text"|"image","page":int,"content":str, "caption":optional}
    for p in tqdm(range(len(doc)), desc="Step 1/6: Extract"):
        page = doc.load_page(p)
        page_num = p + 1

        # TEXT (native -> fallback OCR)
        txt = page.get_text().strip()
        if len(txt) < MIN_NATIVE_CHARS:

            txt = "\n".join(ocr_lines).strip()
        if txt:
            for ch in chunk_text(txt):
                meta.append({"type": "text", "page": page_num, "content": ch})

        # IMAGES
        for idx, img_meta in enumerate(page.get_images(full=True), start=1):
            xref = img_meta[0]
            img = doc.extract_image(xref)
            ext = img.get("ext", "png")
            fname = f"page_{page_num:03d}_img_{idx}.{ext}"

            cap = ""
            if USE_LLAVA_CAPTIONS:

            meta.append({"type": "image", "page": page_num, "content": path, "caption": cap})
    return meta

# ------------------------- Step 2: Vector store (text-only) -------------------------
def build_index(meta):
    qdrant.recreate_collection(
        collection_name="patent_chunks",
        vectors_config=VectorParams(size=embed_dim, distance=Distance.COSINE)
    )
    texts = [m for m in meta if m["type"] == "text"]
    if not texts:
        return 0
    vecs = embedder.encode([t["content"] for t in texts],
                           batch_size=64, convert_to_numpy=True, show_progress_bar=True)
    points = [
        PointStruct(id=i, vector=vecs[i].tolist(),
                    payload={"type":"text","page":texts[i]["page"],"content":texts[i]["content"]})
        for i in range(len(texts))
    ]
    qdrant.upsert(collection_name="patent_chunks", points=points)
    return len(points)

def retrieve_for_question(question, meta):
    qv = embedder.encode(question).tolist()
    res = qdrant.query_points(
        collection_name="patent_chunks",
        query=qv, limit=TOP_K, with_payload=True, score_threshold=MIN_SCORE
    )

    chunks, pages = [], set()
    for pt in res.points:
        c = pt.payload["content"]
        chunks.append(c)
        pages.add(pt.payload["page"])

    # choose up to 2 images from same/adjacent pages
    imgs = []
    img_caps = []
    for pg in sorted(pages):
        if len(imgs) >= MAX_IMAGES: break
        for m in meta:
            if m["type"] == "image" and abs(m["page"] - pg) <= 1:
                if m["content"] not in imgs:
                    imgs.append(m["content"])
                    cap = (m.get("caption") or "")
                    if not cap and USE_LLAVA_CAPTIONS:
                        # fallback: caption now with question hint
                        cap = llava_caption(m["content"], question_hint=question)
                    if cap:
                        img_caps.append(f"[Figure p.{m['page']}] {cap}")
                if len(imgs) >= MAX_IMAGES: break
        if len(imgs) >= MAX_IMAGES: break

    # build context text (text chunks + (optional) image captions), enforce byte cap
    ctx_parts = chunks + img_caps
    ctx = "\n".join(ctx_parts)
    ctx = ctx.encode("utf-8")[:MAX_CTX_BYTES].decode("utf-8", "ignore")

    # Prompt structure required by the assignment
    prompt = f"Question: {question}\n\nContext:\n{ctx}\n\nImages:\n{', '.join(imgs[:MAX_IMAGES])}"
    return prompt

def ask_llama(prompt):
    system_guard = (
        "Answer concisely using ONLY the information in the Context. "
        "If the answer is not in Context, reply exactly: Not found in context."
    )
    full = system_guard + "\n\n" + prompt + "\n\nAnswer:"
    try:
        proc = subprocess.run(
            ["ollama", "run", LLAMA_MODEL],
            input=full, capture_output=True, text=True, encoding="utf-8"
        )
        if proc.returncode != 0:
            return "Not found in context."
        out = proc.stdout.strip()
        return trim_to_limit(out, 300)
    except Exception:
        return "Not found in context."

import numpy as np

def mmr_rerank(query_vec, candidate_payloads, candidate_vecs, top_n=6, lambda_mult=0.7):
    # candidate_payloads: list of payload dicts (with 'content', 'page')
    # candidate_vecs: np.array shape (m, d)
    q = np.asarray(query_vec, dtype=np.float32)
    cand = np.asarray(candidate_vecs, dtype=np.float32)
    # cosine sims
    sims = cand @ (q / (np.linalg.norm(q) + 1e-8))
    selected, selected_idx = [], []
    remaining = set(range(len(candidate_payloads)))

    while len(selected) < min(top_n, len(candidate_payloads)) and remaining:
        if not selected:
            i = int(np.argmax(sims))
            selected.append(candidate_payloads[i])
            selected_idx.append(i)
            remaining.remove(i)
            continue
        # diversity term: max similarity to any selected
        selected_vecs = cand[selected_idx]
        div = selected_vecs @ (cand.T / (np.linalg.norm(cand, axis=1, keepdims=True) + 1e-8))
        div_max = np.max(div, axis=0)
        mmr = lambda_mult * sims - (1 - lambda_mult) * div_max
        mmr[list(selected_idx)] = -1e9  # exclude already chosen
        i = int(np.argmax(mmr))
        selected.append(candidate_payloads[i])
        selected_idx.append(i)
        remaining.remove(i)
    return selected

3) Make the byte cap graceful (no mid-sentence amputations)

Replace your hard slice with boundary-aware trimming:

def trim_bytes(text, byte_limit):
    b = text.encode("utf-8")[:byte_limit]
    s = b.decode("utf-8", "ignore")
    # end at a sensible boundary if we truncated
    if len(text.encode("utf-8")) > byte_limit:
        for p in (". ", ".\n", ";\n", ":\n", "?\n", "!\n"):
            i = s.rfind(p)
            if i >= int(len(s) * 0.6):
                return s[:i+1].strip()
        sp = s.rfind(" ")
        return (s[:sp] if sp > 0 else s).strip()
    return s


Use it on the combined context (text chunks + any figure captions).

    # e.g., "ABSTRACT", "CLAIMS", "What is claimed is:", "FIG.", "DESCRIPTION", "BACKGROUND"

    for m in markers:

    return chunk_text(full_text, max_chars=max_chars)

def build_prompt_for_question(q, client, embedder, meta, collection, max_ctx_bytes=2000,
                              top_k=12, score_th=0.15, max_images=2, use_llava_caps=True):

    q_norm = normalize(q)
    # 1) retrieve reranked text chunks
    initial = client.query_points(collection_name=collection, query=embedder.encode(q_norm).tolist(),
                                  limit=top_k, with_payload=True, score_threshold=score_th)
    payloads, vecs = [], []
    for pt in initial.points:
        payloads.append(pt.payload)               # {'type':'text','content','page'}
        # if you didn't store vectors in Qdrant, compute now:
        vecs.append(embedder.encode(normalize(pt.payload["content"])))
    reranked = mmr_rerank(embedder.encode(q_norm), payloads, np.array(vecs, dtype=np.float32),
                          top_n=6, lambda_mult=0.7)

    # 2) gather pages & assemble text
    pages = {p["page"] for p in reranked}
    text_chunks = [p["content"] for p in reranked]

    # 3) optionally add nearby image captions when relevant
    imgs, caps = [], []
    if wants_figures(q):

        imgs, caps = collect_images_with_captions(pages, meta, question=q,

what does sentence transformer
ChatGPT said:

A Sentence Transformer is a type of neural network model from the sentence-transformers
 library that converts text (sentences, paragraphs, sometimes documents) into dense numerical vectors (embeddings).

Why it matters in your pipeline

Input: A piece of text, e.g.
"What is the role of the LLM in relation to the semantic search engine?"

Output: A fixed-length vector, e.g. [0.12, -0.45, 0.87, ...] (usually 384, 512, or 768 dimensions depending on the model).

Use case: Those vectors let you:

Store text in a vector database (Qdrant, FAISS, etc.).

Perform semantic search (find texts with similar meaning, not just keyword matches).

Build RAG pipelines, clustering, deduplication, etc.

How it works under the hood

Based on BERT / RoBERTa / DistilBERT / MiniLM architectures.

Fine-tuned specifically so that semantically similar sentences land close together in vector space.

Compared using similarity metrics (cosine similarity, dot product).

Example
from sentence_transformers import SentenceTransformer

model = SentenceTransformer("all-MiniLM-L6-v2")
emb1 = model.encode("The tank is moving fast.")
emb2 = model.encode("A vehicle is driving quickly.")
emb3 = model.encode("Bananas are yellow.")

print(cosine_similarity([emb1], [emb2]))  # high similarity (~0.8)
print(cosine_similarity([emb1], [emb3]))  # low similarity (~0.1)


So, in your patent pipeline, the Sentence Transformer:

Reads text chunks from the PDF.

Produces embeddings (size = 384 for MiniLM, for example).

Lets Qdrant index them.

Makes it possible to query with a user question → find semantically closest chunks.
